//! overlay rebuilds AtlasData's events/places/narratives/verses FROM THE

use atlas_core::data::AtlasData;
use atlas_graph::service::GraphService;

fn overlay_path() -> AtlasData {
    // EXACTLY the server's default startup (atlas-server/src/load.rs:103-108):
    let gs = GraphService::from_artifact(std::path::Path::new("../../data/compiled/graph.bin")).expect("artifact");
    let mut data = AtlasData::load(std::path::Path::new("../../data/compiled")).expect("sidecars");
    let f = atlas_graph::legacy::atlas_data_overlay(&gs);
    data.events = f.events; data.places = f.places; data.narratives = f.narratives; data.verses = f.verses;
    data.finish();
    data
}
fn sources_path() -> AtlasData {
    // The compile binary's own input (bins/compile_graph.rs builds its graph from this):
    let mut data = atlas_etl::compile::compile(std::path::Path::new("../../data/raw"), std::path::Path::new("../../data/curated")).expect("compile");
    data.finish();
    data
}

#[test]
fn overlay_and_sources_agree_on_events_places_narratives_and_verses() {
    let a = overlay_path(); let b = sources_path();
    assert_eq!(a.events.len(), b.events.len(), "event count");
    for (x, y) in a.events.iter().zip(&b.events) {
        assert_eq!(x.id, y.id, "event order differs at {}", x.id);
        assert_eq!(x.places, y.places, "places (incl. anchor order) for {}", x.id);
        assert_eq!(x.when, y.when, "when for {}", x.id);
        assert_eq!(x.verses, y.verses, "verses for {}", x.id);
        assert_eq!(x.label, y.label, "label for {}", x.id);
    }
    assert_eq!(a.places.len(), b.places.len());
    for (x, y) in a.places.iter().zip(&b.places) { assert_eq!(x.id, y.id); assert_eq!((x.lat, x.lon, &x.name, &x.verse_links), (y.lat, y.lon, &y.name, &y.verse_links), "place {}", x.id); }
    assert_eq!(a.narratives, b.narratives);
    assert_eq!(a.verses.len(), b.verses.len(), "KJV verse count");
    assert_eq!(a.verses, b.verses, "verse text");
}
