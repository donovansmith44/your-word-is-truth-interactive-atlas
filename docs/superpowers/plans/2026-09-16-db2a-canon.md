```rust
// graph-types/src/canon/mod.rs
pub const CANON_VERSION: u32 = 1;
pub const DOMAIN_PREFIX: &[u8] = b"bible-atlas/canon/1\n";

#[derive(Clone, Debug, PartialEq)]
pub enum Value { Null, Bool(bool), Int(i64), Float(f64), Str(String), Arr(Vec<Value>), Obj(std::collections::BTreeMap<String, Value>) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CanonError { pub path: String, pub msg: String }   // path like "payload.Place.lat"

pub fn serialize(v: &Value) -> Vec<u8>;                       // canonical bytes, never fails (floats validated at Value construction: `Value::float(f64) -> Result<Value, CanonError>`)
pub fn parse(bytes: &[u8]) -> Result<Value, CanonError>;      // strict: rejects whitespace, duplicate keys, unsorted keys, non-minimal escapes, leading zeros, trailing content

pub trait Canon: Sized {
    fn to_value(&self) -> Value;
    fn from_value(v: &Value) -> Result<Self, CanonError>;
    fn encode(&self) -> Vec<u8> { serialize(&self.to_value()) }
    fn decode(bytes: &[u8]) -> Result<Self, CanonError> { Self::from_value(&parse(bytes)?) }
}

// graph-types/src/canon/ids.rs
pub fn node_kind_str(k: NodeKind) -> &'static str;             // "TextUnit" … "CommentaryItem" (the Debug names)
pub fn parse_node_kind(s: &str) -> Result<NodeKind, CanonError>;
pub fn any_node_id_str(id: &AnyNodeId) -> String;             // "Place:jerusalem", "TextUnit:bible/JHN.3.16" — kind, ':', raw (raw may contain ':' — split at the FIRST ':')
pub fn parse_any_node_id(s: &str) -> Result<AnyNodeId, CanonError>;
pub fn position_str(p: &Position) -> String;                  // "n:" + any_node_id_str | "e:" + edge_id.0
pub fn parse_position(s: &str) -> Result<Position, CanonError>;
```
